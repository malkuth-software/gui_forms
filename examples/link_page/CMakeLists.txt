cmake_minimum_required(VERSION 3.25)
project(GUIFormsLinkPageExample LANGUAGES CXX)
option(GUI_FORMS_EXAMPLE_NATIVE "Open the page in a native window" OFF)
if(GUI_FORMS_EXAMPLE_NATIVE)
    find_package(GUIForms 1.0 CONFIG REQUIRED COMPONENTS Application)
else()
    find_package(GUIForms 1.0 CONFIG REQUIRED COMPONENTS Controls)
endif()
add_executable(link_page main.cpp link_page.cpp sample_document.cpp)
target_compile_features(link_page PRIVATE cxx_std_20)
if(GUI_FORMS_EXAMPLE_NATIVE)
    target_compile_definitions(link_page PRIVATE GUI_FORMS_EXAMPLE_NATIVE=1)
    target_link_libraries(link_page PRIVATE GUIForms::Application)
else()
    target_link_libraries(link_page PRIVATE GUIForms::Controls)
endif()
enable_testing()
add_test(NAME link_page COMMAND link_page)

# Each program's window is a form document (forms/), generated at build time
# into a gui_forms::Window subclass; a model header beside the program names
# what the document binds.
function(reference_window target document class)
    cmake_parse_arguments(PARSE_ARGV 3 WINDOW "" "MODEL;MODEL_HEADER" "")
    if(WINDOW_MODEL)
        gui_forms_add_form(${target} ${document} CLASS ${class} NAMESPACE reference
                           MODEL ${WINDOW_MODEL} MODEL_HEADER ${WINDOW_MODEL_HEADER})
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    else()
        gui_forms_add_form(${target} ${document} CLASS ${class} NAMESPACE reference DERIVED)
    endif()
endfunction()

    add_executable(${example_name} ${example_name}.cpp)
    target_compile_features(${example_name} PRIVATE cxx_std_20)

        target_link_libraries(${example_name} PRIVATE GUIForms::Threading)

    if(example_name STREQUAL "bound_values")
        reference_window(${example_name} forms/bound_values.form SettingsWindow)
    elseif(example_name STREQUAL "shared_state")

                         MODEL reference::SharedModel MODEL_HEADER shared_state_model.hpp)
    elseif(example_name STREQUAL "collection_state")
        reference_window(${example_name} forms/collection_state.form CollectionsWindow)
    elseif(example_name STREQUAL "multiline_scrollbars")
        reference_window(${example_name} forms/multiline_scrollbars.form ViewerWindow)
    elseif(example_name STREQUAL "window_rules")
        reference_window(${example_name} forms/window_rules.form RulesWindow)
    elseif(example_name STREQUAL "mouse_events")
        reference_window(${example_name} forms/mouse_events.form MouseWindow)
    endif()
    add_test(NAME ${example_name} COMMAND ${example_name})
    set_tests_properties(${example_name} PROPERTIES TIMEOUT 15)
endforeach()

# FD gate item 3 (owner 2026-11-08, option 1): every Gallery page document
# generated by an installed form compiler, and each generated header
# compiled against the installed headers. The page documents are read from
# this source tree (demo/pages); the Gallery model itself is not linked. A
# page's header is the document class template over an owner, so the
# compile checks the header and every name it fixes; the owner's members are
# checked where the Gallery instantiates it. The test prints the number of
# documents, which the gate record notes.
#
# Some pages name the lab controls (CorrespondenceView, ReviewCard,
# InstrumentRack) and the demonstration controls (EasingPreview,
# TypographySpecimen, VisualInspectorView/Overlay), which left the installed
# library (audit F9). Those documents are generated by the Gallery's form
# compiler (GUIForms::form_compiler_lab) and compiled against GUIForms::Lab
# and GUIForms::DemoControls, which an SDK installs only with
# GUI_FORMS_INSTALL_LAB and GUI_FORMS_INSTALL_GALLERY_MODEL. Without them the
# configure fails: no page is skipped.
file(GLOB gallery_page_forms CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../demo/pages/*.form")
if(gallery_page_forms AND TARGET GUIForms::form_compiler)
    set(gallery_page_missing "")
    foreach(gallery_page_target IN ITEMS GUIForms::form_compiler_lab GUIForms::Lab GUIForms::DemoControls)
        if(NOT TARGET ${gallery_page_target})
            list(APPEND gallery_page_missing ${gallery_page_target})
        endif()
    endforeach()
    if(gallery_page_missing)
        list(JOIN gallery_page_missing ", " gallery_page_missing_text)
        message(FATAL_ERROR "installed_gallery_page_forms: the SDK has no ${gallery_page_missing_text}. "
                            "The Gallery pages name the lab and demonstration controls (audit F9): install "
                            "GUI.Forms with GUI_FORMS_INSTALL_LAB=ON and GUI_FORMS_INSTALL_GALLERY_MODEL=ON.")
    endif()
    list(LENGTH gallery_page_forms gallery_page_form_count)
    set(gallery_page_includes "")
    set(gallery_page_source "${CMAKE_CURRENT_BINARY_DIR}/installed_gallery_page_forms.cpp")
    add_executable(installed_gallery_page_forms "${gallery_page_source}")
    target_compile_features(installed_gallery_page_forms PRIVATE cxx_std_20)
    target_link_libraries(installed_gallery_page_forms PRIVATE
        GUIForms::Controls GUIForms::Lab GUIForms::DemoControls)
    set_target_properties(installed_gallery_page_forms PROPERTIES NO_SYSTEM_FROM_IMPORTED TRUE)
    foreach(page_form IN LISTS gallery_page_forms)
        get_filename_component(page_stem "${page_form}" NAME_WE)
        string(MAKE_C_IDENTIFIER "${page_stem}" page_identifier)
        # Every page through the Gallery's compiler: it knows the registry's
        # types and the lab and demonstration controls.
        gui_forms_add_form(installed_gallery_page_forms "${page_form}"
            CLASS ${page_identifier}_document NAMESPACE installed_gallery_pages
            COMPILER GUIForms::form_compiler_lab)
        string(APPEND gallery_page_includes "#include \"${page_stem}_form.hpp\"\n")
    endforeach()
    # Written only when its bytes change (file(CONFIGURE)), so a configure
    # rebuilds nothing that did not change.
    file(CONFIGURE OUTPUT "${gallery_page_source}" CONTENT
"// Generated by examples/reference/CMakeLists.txt: every Gallery page document,
// generated by the installed form compiler, included once.
${gallery_page_includes}
#include <iostream>

int main() {
    std::cout << \"installed gallery page forms: ${gallery_page_form_count}\\n\";
    return 0;
}
" @ONLY)
    add_test(NAME installed_gallery_page_forms COMMAND installed_gallery_page_forms)
    set_tests_properties(installed_gallery_page_forms PROPERTIES TIMEOUT 15
        PASS_REGULAR_EXPRESSION "installed gallery page forms: ${gallery_page_form_count}")
elseif(gallery_page_forms)
    message(STATUS "installed_gallery_page_forms: skipped, the SDK has no form compiler")
endif()

# Installed Windows SDKs keep DLLs in bin. Stage the actual transitive runtime
# targets alongside each executable; no Core archive is copied into consumers.
if(WIN32)
    get_property(reference_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(reference_target IN LISTS reference_targets)
        get_target_property(reference_type ${reference_target} TYPE)
        if(reference_type STREQUAL "EXECUTABLE")
            add_custom_command(TARGET ${reference_target} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    $<TARGET_RUNTIME_DLLS:${reference_target}>
                    $<TARGET_FILE_DIR:${reference_target}>
                COMMAND_EXPAND_LISTS)
        endif()
    endforeach()
endif()

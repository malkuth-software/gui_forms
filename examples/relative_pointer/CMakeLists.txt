cmake_minimum_required(VERSION 3.25)
project(GUIFormsRelativePointerExample LANGUAGES CXX)
option(GUI_FORMS_EXAMPLE_NATIVE "Run the orbit toy in a native window" OFF)
if(GUI_FORMS_EXAMPLE_NATIVE)
    find_package(GUIForms 1.0 CONFIG REQUIRED COMPONENTS Application)
else()
    find_package(GUIForms 1.0 CONFIG REQUIRED COMPONENTS Controls)
endif()
add_executable(orbit_toy main.cpp orbit_toy.cpp)
target_compile_features(orbit_toy PRIVATE cxx_std_20)
if(GUI_FORMS_EXAMPLE_NATIVE)
    target_compile_definitions(orbit_toy PRIVATE GUI_FORMS_EXAMPLE_NATIVE=1)
    target_link_libraries(orbit_toy PRIVATE GUIForms::Application)
else()
    target_link_libraries(orbit_toy PRIVATE GUIForms::Controls)
endif()
enable_testing()
add_test(NAME orbit_toy COMMAND orbit_toy)
